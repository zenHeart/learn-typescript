Rollup 打包流程

// 打包流程 Input Files → Plugins → Module Graph → Bundle → Output Plugins → Output Files // 核心优势 1. ES Module 原生支持，Tree-shaking 效率高 2. 静态分析，可生成更小的包体积 3. 配置文件简洁，学习成本低 4. 输出格式丰富，适配多场景

快速开始

# 安装核心依赖 npm install rollup --save-dev # 安装 TypeScript 相关插件 npm install @rollup/plugin-typescript @rollup/plugin-node-resolve @rollup/plugin-commonjs --save-dev # 创建 rollup.config.js export default { input: 'src/index.ts', output: { format: 'es', file: 'dist/bundle.mjs' }, plugins: [ /* 插件配置 */ ] }; # 执行打包 npx rollup -c

四大核心插件

@rollup/plugin-typescript
TypeScript 编译插件，将 .ts/.tsx 文件编译为 JavaScript，支持声明文件生成。
@rollup/plugin-node-resolve
模块解析插件，帮助 Rollup 查找 node_modules 中的第三方模块。
@rollup/plugin-commonjs
CommonJS 转换插件，将 CommonJS 模块转换为 ES Module。
@rollup/plugin-terser
压缩插件，生产环境使用，去除冗余代码和注释。

插件配置示例

import typescript from '@rollup/plugin-typescript'; import resolve from '@rollup/plugin-node-resolve'; import commonjs from '@rollup/plugin-commonjs'; import terser from '@rollup/plugin-terser'; const production = !process.env.ROLLUP_WATCH; export default { plugins: [ // TypeScript 编译 typescript({ tsconfig: './tsconfig.json', declaration: true, declarationDir: './dist/types' }), // 解析 node_modules resolve({ extensions: ['.js', '.ts', '.jsx', '.tsx'], preferBuiltins: false }), // CommonJS 转换 commonjs({ transformMixedEsModules: true }), // 生产环境压缩 production && terser({ compress: { drop_console: true } }) ] };

tsconfig.json 推荐配置

{ "compilerOptions": { // 关键：保持 ES Module 特性 "target": "ESNext", "module": "ESNext", "moduleResolution": "node", "strict": true, // 声明文件配置 "declaration": true, "declarationDir": "./dist/types", "sourceMap": true, // ES Module 兼容 "esModuleInterop": true, "allowSyntheticDefaultImports": true, // 跳过 node_modules 检查 "skipLibCheck": true }, "include": ["src/**/*"], "exclude": ["node_modules", "dist"] }

关键配置说明

// 1. target: "ESNext" // 让 TypeScript 编译为最新的 JavaScript 语法 // 保留 ES Module 的 import/export 语句 // 2. module: "ESNext" // Rollup 需要 ES Module 格式进行 Tree-shaking // 不要使用 "CommonJS"，会丢失 Tree-shaking 能力 // 3. declaration: true // 生成 .d.ts 声明文件 // 库使用者需要类型提示 // 4. skipLibCheck: true // 跳过 node_modules 的类型检查 // 加快编译速度

四种输出格式对比

ES Module
.mjs / .esm.js
现代浏览器和打包工具首选，支持 Tree-shaking，适合 npm package 的 main 字段。
CommonJS
.cjs / .js
Node.js 环境使用，require() 语法，适合 package.json 的 main 字段。
UMD
.umd.js
通用格式，支持 AMD/CommonJS/全局变量，适合需要同时支持多环境的库。
IIFE
.iife.js
自执行函数，直接运行于浏览器，通过 script 标签引入。

多格式输出配置

export default { input: 'src/index.ts', // 多格式输出 output: [ // ES Module { format: 'es', file: 'dist/index.mjs', sourcemap: true }, // CommonJS { format: 'cjs', file: 'dist/index.cjs.js', sourcemap: true }, // UMD { format: 'umd', file: 'dist/index.umd.js', name: 'MyLibrary', globals: { lodash: '_' // 外部依赖 } } ] };

完整项目配置

// rollup.config.js import resolve from '@rollup/plugin-node-resolve'; import commonjs from '@rollup/plugin-commonjs'; import typescript from '@rollup/plugin-typescript'; import terser from '@rollup/plugin-terser'; const production = !process.env.ROLLUP_WATCH; export default { input: 'src/index.ts', external: ['lodash'], plugins: [ typescript({ tsconfig: './tsconfig.json', declaration: true, declarationDir: './dist/types' }), resolve({ extensions: ['.js', '.ts'] }), commonjs(), production && terser() ], output: [ { format: 'es', file: 'dist/index.mjs' }, { format: 'cjs', file: 'dist/index.cjs.js' }, { format: 'umd', file: 'dist/index.umd.js', name: 'MyLibrary', globals: { lodash: '_' } } ] };

package.json 配置

{ "name": "my-library", "version": "1.0.0", "type": "module", // 多格式入口 "main": "dist/index.cjs.js", "module": "dist/index.mjs", "types": "dist/types/index.d.ts", "exports": { ".": { "import": "dist/index.mjs", "require": "dist/index.cjs.js", "types": "dist/types/index.d.ts" } }, "scripts": { "build": "rollup -c", "build:watch": "rollup -c -w" } }
